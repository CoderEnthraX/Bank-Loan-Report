// accuracy of the shown trades, credited when a trade closes. A target counts as reached when the best price after the fill
// touched it before the stop (a bar that also hits the stop adds nothing: the order inside a bar is unknown); the final target
// is reached exactly when the trade is a win.
var array<int> tpHitN = array.new_int(4, 0)
var int beN = 0      // closed at a stop moved to the entry (breakeven, 0R)
var int lkN = 0      // closed at a stop moved into profit (locked profit)
if evExit
    float sd = lastDir == 1 ? 1.0 : -1.0
    for k = 1 to 4
        float lvl = k == 1 ? t1 : k == 2 ? t2 : k == 3 ? t3 : t4
        if evWin ? k <= exitK : (k < exitK and sd * (bestPx - lvl) >= -syminfo.mintick * 0.001)
            array.set(tpHitN, k - 1, array.get(tpHitN, k - 1) + 1)
    if not evWin and stp != stp0
        if evRes == 0.0
            beN += 1
        else if evRes > 0.0
            lkN += 1
bool mvAny  = mv1On or mv2On or mv3On
bool fltAny = noChase or chopOn or emaOn
f_sgnR(float r) => (r >= 0 ? "+" : "") + str.tostring(r, "0.00") + "R"
f_share(int n, int total) => total > 0 ? str.tostring(math.round(100.0 * n / total)) + "%" : "—"
// "77 trades · 52% (40W/37L) · +12.00R"
f_accTxt() => str.tostring(totW + totL) + " trades · " + f_share(totW, totW + totL) + " (" + str.tostring(totW) + "W/" + str.tostring(totL) + "L) · " + f_sgnR(totR)
// "TP1 61% (47) · TP2 49% (38) · TP3 40% (31) · TP4 34% (26)"
f_tpTxt() =>
    string s = ""
    for k = 1 to 4
        int n = array.get(tpHitN, k - 1)
        s := s + (k > 1 ? " · " : "") + "TP" + str.tostring(k) + " " + (k > exitK ? "n/a" : f_share(n, totW + totL) + " (" + str.tostring(n) + ")")
    s
f_beTxt() => "BE " + str.tostring(beN) + " (" + f_share(beN, totW + totL) + ") · locked profit " + str.tostring(lkN) + " (" + f_share(lkN, totW + totL) + ")"
f_fltTxt() => str.tostring(ghW + ghL) + " skipped · would have been " + str.tostring(ghW) + "W/" + str.tostring(ghL) + "L · " + f_sgnR(ghR)
