// >>> ENGINE BEGIN (shared: byte-identical in Auction_Retest_Model.pine and Auction_Retest_Strategy.pine)
// The engine decides every setup, fill, stop move and exit from the chart's bars alone. The strategy only copies its decisions
// into orders, so the indicator and the strategy always show the same signals and trades.

// ── session VWAP (volume weighted; equal weights when the symbol has no volume) and its standard deviation ──
bool  chgD      = timeframe.change("D")
bool  chgW      = timeframe.change("W")
bool  chgM      = timeframe.change("M")
bool  newAnchor = vwapAnchor == "Week" ? chgW : vwapAnchor == "Month" ? chgM : chgD
bool  newDay    = chgD
bool  hasVolume = ta.cum(nz(volume)) > 0
float vw_w      = hasVolume ? nz(volume) : 1.0
var float cumPV  = 0.0
var float cumPV2 = 0.0
var float cumW   = 0.0
var float cumP   = 0.0
var float cumP2  = 0.0
var int   cumN   = 0
var int   vwBars = 0      // bars since the VWAP period started (0 on its first bar)
var int   sessN  = 0      // completed VWAP periods on the chart (the first one is usually partial)
if newAnchor
    cumPV  := 0.0
    cumPV2 := 0.0
    cumW   := 0.0
    cumP   := 0.0
    cumP2  := 0.0
    cumN   := 0
    vwBars := 0
    if bar_index > 0
        sessN += 1
else
    vwBars += 1
cumPV  += vwapSrc * vw_w
cumPV2 += vwapSrc * vwapSrc * vw_w
cumW   += vw_w
cumP   += vwapSrc
cumP2  += vwapSrc * vwapSrc
cumN   += 1
float vwap  = cumW > 0 ? cumPV / cumW : cumP / cumN
float vwVar = cumW > 0 ? cumPV2 / cumW - vwap * vwap : cumP2 / cumN - vwap * vwap
float vwSd  = math.sqrt(math.max(vwVar, 0.0))

// ── market structure and the displacement candle ──
float atr       = ta.atr(atrLen)
// warm-up: no impulses are counted and no setup is armed until every lookback and the ATR exist
bool  warm      = bar_index >= math.max(structLen, brkLen, entryLen, momLen, slopeBars, stopLen) + 1 and not na(atr)
float priorHigh = ta.highest(high, structLen)[1]
float priorLow  = ta.lowest(low, structLen)[1]
float sigHigh   = ta.highest(high, brkLen)[1]
float sigLow    = ta.lowest(low, brkLen)[1]
float entHigh   = ta.highest(high, math.min(entryLen, structLen))[1]
float entLow    = ta.lowest(low, math.min(entryLen, structLen))[1]
float swingLow  = ta.lowest(low, stopLen)
float swingHigh = ta.highest(high, stopLen)
float rng       = high - low
float body      = math.abs(close - open)
float bodyMid   = (open + close) / 2.0
float momMove   = close - close[momLen]
float erDen     = math.sum(math.abs(close - close[1]), momLen)
bool  erUp      = cmoMin <= 0 or (erDen > 0 and momMove / erDen >= cmoMin / 100.0)
bool  erDn      = cmoMin <= 0 or (erDen > 0 and -momMove / erDen >= cmoMin / 100.0)
float vwSlope   = slopeBars > 0 and vwBars >= slopeBars ? vwap - vwap[slopeBars] : na
bool  slopeUp   = na(vwSlope) or vwSlope > 0
bool  slopeDn   = na(vwSlope) or vwSlope < 0
float sgSlope   = slopeBars <= 0 ? na : slopeCross ? vwap - vwap[slopeBars] : vwSlope
bool  sigUp     = na(sgSlope) or sgSlope > 0
bool  sigDn     = na(sgSlope) or sgSlope < 0
// percentage thresholds accept an exact tie (11/20 ticks = 55 %), whatever the float rounding
bool  bigCandle = rng > 0 and rng >= dispAtr * atr and body / rng * 100.0 >= dispBodyPct - 1e-7
bool  bigImp    = rng > 0 and rng >= dispAtr * atr and body / rng * 100.0 >= impBodyPct - 1e-7
bool  closeHi   = rng > 0 and (high - close) / rng * 100.0 <= dispClosePct + 1e-7
bool  closeLo   = rng > 0 and (close - low) / rng * 100.0 <= dispClosePct + 1e-7
bool  bullImp   = close > open and high > priorHigh and bigImp and closeHi and slopeUp
bool  bearImp   = close < open and low < priorLow and bigImp and closeLo and slopeDn
bool  setupOk   = vwBars >= minOpenBars and not (skipSunday and dayofweek(time, "UTC") == dayofweek.sunday)
// every displacement condition except the structure break, which is tested in step 3 (it can be re-anchored after a trade)
bool  bullBase  = allowLong and setupOk and close > open and bigCandle and closeHi and sigUp and (not vwapFilter or close > vwap) and (momAtr <= 0 or momMove >= momAtr * atr) and erUp
bool  bearBase  = allowShort and setupOk and close < open and bigCandle and closeLo and sigDn and (not vwapFilter or close < vwap) and (momAtr <= 0 or -momMove >= momAtr * atr) and erDn
// optional deep-pullback signal: breaks the signal level while still well inside the structure range
float momB      = close - close[brkLen]
float erDenB    = math.sum(math.abs(close - close[1]), brkLen)
bool  deepL     = deepOn and allowLong and setupOk and close > open and bigCandle and closeHi and sigUp and close > vwap and close > sigHigh and momB >= momAtr * atr and erDenB > 0 and momB / erDenB >= cmoMin / 100.0 and priorHigh - close >= deepFar * atr
bool  deepS     = deepOn and allowShort and setupOk and close < open and bigCandle and closeLo and sigDn and close < vwap and close < sigLow and -momB >= momAtr * atr and erDenB > 0 and -momB / erDenB >= cmoMin / 100.0 and close - priorLow >= deepFar * atr

// ── optional accuracy filters (off by default) ──
// VWAP crosses in this VWAP period: the close moving from more than 0.2 ATR on one side of the VWAP to more than 0.2 ATR on the other
var int vxSide = 0
var int vxN    = 0
if newAnchor
    vxSide := 0
    vxN    := 0
float vxDev  = not na(atr) and atr > 0 ? (close - vwap) / atr : 0.0
int   vxZ    = vxDev > 0.2 ? 1 : vxDev < -0.2 ? -1 : 0
if vxZ != 0
    if vxSide != 0 and vxZ != vxSide
        vxN += 1
    vxSide := vxZ
float vxRate = 100.0 * vxN / (vwBars + 1)
// 1-hour candles (UTC hours) built from the chart's bars and their EMA20; hE0..hE3 = the EMA of the last four COMPLETED hours
var int   hK   = na
var float hC   = na
var int   hNum = 0
var float hSum = 0.0
var float hE0  = na
var float hE1  = na
var float hE2  = na
var float hE3  = na
int hKey = math.floor(time / 3600000.0)
if not na(hK) and hKey != hK
    // the previous hour is complete; the EMA starts from the average of the first 20 hours
    hNum += 1
    float hNew = na
    if hNum <= 20
        hSum += hC
        hNew := hNum == 20 ? hSum / 20.0 : na
    else
        hNew := 2.0 / 21.0 * hC + (1.0 - 2.0 / 21.0) * hE0
    hE3 := hE2
    hE2 := hE1
    hE1 := hE0
    hE0 := hNew
hK := hKey
hC := close
float emaSpd = not na(hE0) and not na(hE3) and not na(atr) and atr > 0 ? math.abs(hE0 - hE3) / atr : na

// ── session counters: day stats, impulses for the half stop, session range ──
var int   dayFills   = 0
var int   dayW       = 0
var int   dayL       = 0
var float dayR       = 0.0
var float dayRiskR   = 0.0     // R counted toward the daily stop
var int   lossDirBlk = 0       // direction of the last loss this session (accuracy filter)
var int   bullN      = 0
var int   bearN      = 0
var int   prevNet    = na
var float sesO       = na
var float sesH       = na
var float sesL       = na
var float pdH        = na
var float pdL        = na
if newDay
    dayFills   := 0
    dayW       := 0
    dayL       := 0
    dayR       := 0.0
    dayRiskR   := 0.0
    lossDirBlk := 0
    if bar_index > 0
        prevNet := bullN - bearN
    bullN := 0
    bearN := 0
    pdH   := sesH
    pdL   := sesL
    sesH  := high
    sesL  := low
    sesO  := open
else
    sesH := na(sesH) ? high : math.max(sesH, high)
    sesL := na(sesL) ? low : math.min(sesL, low)
// half stop: the first impulse of a session against a strongly opposite previous session, or after a gap outside its range
bool halfL = halfStopOn and bullN == 0 and not na(prevNet) and prevNet <= -biasNet
bool halfS = halfStopOn and bearN == 0 and not na(prevNet) and prevNet >= biasNet
if halfStrict and not na(prevNet)
    halfL := halfL and (prevNet <= -5 or (prevNet + bullN - bearN <= -1 and bearN >= 1))
    halfS := halfS and (prevNet >= 5 or (prevNet + bullN - bearN >= 1 and bullN >= 1))
if halfOpenOut and not na(pdH)
    halfL := halfL or (halfStopOn and bullN == 0 and sesO < pdL)
    halfS := halfS or (halfStopOn and bearN == 0 and sesO > pdH)
if bullImp and warm
    bullN += 1
if bearImp and warm
    bearN += 1

// ── model state ── st: 0 scanning · 1 buy armed · -1 sell armed · 2 long active · -2 short active
var int   st       = 0
var int   armBar   = na
var int   fillBar  = na
var int   lastExit = na
var int   lastFill = na      // fill bar and direction of the most recent trade (re-anchor)
var int   lastDir  = 0
var float lim      = na      // entry (unrounded, like the reference)
var float stp      = na      // current stop
var float stp0     = na      // initial stop
var float trRisk   = na      // entry-to-initial-stop distance
var float t1       = na
var float t2       = na
var float t3       = na
var float t4       = na
var float tF       = na      // final target (Final Exit At)
var int   tpHit    = 0
var bool  trHalf   = false   // the armed / live setup uses the half stop
var bool  trGhost  = false   // skipped by an accuracy filter: runs invisibly, never shown or traded
var string trSkip  = ""      // which filter skipped it
var int   totW     = 0
var int   totL     = 0
var float totR     = 0.0
var int   ghW      = 0       // results the skipped (invisible) trades would have had
var int   ghL      = 0
var float ghR      = 0.0
var string lastRes = "—"
var float bestPx   = na
// events of this bar (shown setups / trades only)
bool   evArm     = false
bool   evArmL    = false
bool   evFill    = false
bool   evFillL   = false
bool   evExit    = false
bool   evWin     = false
float  evRes     = na
bool   evMove    = false
bool   evTp      = false
string evCancel  = ""

// ── 1. manage the live trade: stop first, then the final target, else stop moves (from the next bar) ──
if math.abs(st) == 2 and bar_index > fillBar
    bool  isL     = st == 2
    float d       = isL ? 1.0 : -1.0
    // a print exactly at a level counts (the tolerance only absorbs float noise of the unrounded levels)
    bool  stopHit = isL ? low <= stp + syminfo.mintick * 0.001 : high >= stp - syminfo.mintick * 0.001
    bool  tgtHit  = isL ? high >= tF - syminfo.mintick * 0.001 : low <= tF + syminfo.mintick * 0.001
    int   reached = (isL ? high >= t3 : low <= t3) ? 3 : (isL ? high >= t2 : low <= t2) ? 2 : (isL ? high >= t1 : low <= t1) ? 1 : 0
    int   prevHit = tpHit
    tpHit := math.max(tpHit, reached)
    // best price after the fill; the fill bar and a stop bar add nothing (order inside a bar is unknown)
    if not stopHit
        bestPx := isL ? math.max(bestPx, high) : math.min(bestPx, low)
    if stopHit or tgtHit
        // a bar touching both counts as the stop
        bool  win = not stopHit
        float res = win ? exitR : stp == stp0 ? -1.0 : d * (stp - lim) / trRisk
        if not trGhost
            dayRiskR   += res
            lossDirBlk := res < 0 ? (isL ? 1 : -1) : res > 0 ? 0 : lossDirBlk
            totR       += res
            dayR       += res
            totW       += win ? 1 : 0
            dayW       += win ? 1 : 0
            totL       += win ? 0 : 1
            dayL       += win ? 0 : 1
            lastRes    := (win ? "TP" + str.tostring(exitK) + " " : "STOP ") + (res >= 0 ? "+" : "") + str.tostring(res, "0.00") + "R"
            evExit     := true
            evWin      := win
            evRes      := res
        else
            ghW += win ? 1 : 0
            ghL += win ? 0 : 1
            ghR += res
        lastExit := bar_index
        st       := 0
    else if tpHit > prevHit
        float ns = stp
        if mv1On and tpHit >= 1
            ns := isL ? math.max(ns, lim + mv1R * trRisk) : math.min(ns, lim - mv1R * trRisk)
        if mv2On and tpHit >= 2
            ns := isL ? math.max(ns, lim + mv2R * trRisk) : math.min(ns, lim - mv2R * trRisk)
        if mv3On and tpHit >= 3
            ns := isL ? math.max(ns, lim + mv3R * trRisk) : math.min(ns, lim - mv3R * trRisk)
        if ns != stp
            stp    := ns
            evMove := not trGhost
        evTp := not trGhost

// ── 2. manage the armed setup: fill (limit touch, or the next bar's open for market entries) or expiry ──
if math.abs(st) == 1 and bar_index > armBar
    bool isL     = st == 1
    // a print exactly at the limit fills (the tolerance only absorbs float noise of the unrounded level)
    bool touched = entryMode == E_MARKET ? true : isL ? low <= lim + syminfo.mintick * 0.001 : high >= lim - syminfo.mintick * 0.001
    if touched and dayFills < maxFills
        st       := isL ? 2 : -2
        fillBar  := bar_index
        lastFill := bar_index
        lastDir  := isL ? 1 : -1
        tpHit    := 0
        dayFills += 1
        bestPx   := lim
        if not trGhost
            evFill  := true
            evFillL := isL
        // the fill bar also trading through the stop is an immediate loss
        if isL ? low <= stp + syminfo.mintick * 0.001 : high >= stp - syminfo.mintick * 0.001
            if not trGhost
                dayRiskR   += fillBarStopR ? -1.0 : 0.0
                lossDirBlk := isL ? 1 : -1
                totR       += -1.0
                dayR       += -1.0
                totL       += 1
                dayL       += 1
                lastRes    := "STOP -1.00R"
                evExit     := true
                evWin      := false
                evRes      := -1.0
            else
                ghL += 1
                ghR += -1.0
            lastExit := bar_index
            st       := 0
    else if bar_index - armBar >= retestBars and barstate.isconfirmed
        // expiry is final only at the close: on the live bar the limit can still fill until then
        st := 0
        if not trGhost
            evCancel := "EXPIRED"

// ── 3. arm a new setup on a confirmed displacement candle ──
// no new setup on the exit bar itself, then the cooldown
bool  cooled  = na(lastExit) or bar_index - lastExit >= math.max(cooldownBars, 1)
bool  dayOpen = dayStopR <= 0 or dayRiskR > -dayStopR
// structure break: the signal-break extreme, or shortly after a closed trade the extreme since its fill (same direction only)
int   legN    = na(lastFill) ? 0 : bar_index - lastFill
bool  legOn   = legReset and math.abs(st) != 2 and legN > 0 and legN < structLen
float brkHigh = sigHigh
float brkLow  = sigLow
if legOn
    float legHigh = high[1]
    float legLow  = low[1]
    for j = 1 to legN
        legHigh := math.max(legHigh, high[j])
        legLow  := math.min(legLow, low[j])
    if lastDir == 1
        brkHigh := legHigh
    else
        brkLow := legLow
bool bullDisp = (bullBase and close > brkHigh) or deepL
bool bearDisp = (bearBase and close < brkLow) or deepS
if lossBlock and lossDirBlk == 1
    bullDisp := false
if lossBlock and lossDirBlk == -1
    bearDisp := false
int  dowUse = dayofweek(time, tzUse)
bool dayOk  = not dayOn or (dowUse == dayofweek.monday and dMon) or (dowUse == dayofweek.tuesday and dTue) or (dowUse == dayofweek.wednesday and dWed) or (dowUse == dayofweek.thursday and dThu) or (dowUse == dayofweek.friday and dFri) or (dowUse == dayofweek.saturday and dSat) or (dowUse == dayofweek.sunday and dSun)
bool inHours = not na(time(timeframe.period, hoursSess, tzUse))
bool hourOk  = not hoursOn or inHours
bool canArm  = barstate.isconfirmed and warm and dayOk and hourOk and math.abs(st) != 2 and cooled and dayFills < maxFills and dayOpen and atr > 0
bool keepArm = rearmMode == REARM_KEEP and ((st == 1 and bullDisp) or (st == -1 and bearDisp))
if canArm and (bullDisp or bearDisp) and not keepArm
    bool  isL   = bullDisp
    float d     = isL ? 1.0 : -1.0
    float entry = entryMode == E_MODEL ? (isL ? math.max(bodyMid, entHigh) : math.min(bodyMid, entLow)) : entryMode == E_PCT ? close - d * entryPct / 100.0 * rng : close
    bool  half  = stopMode == STOP_ATR and (isL ? halfL : halfS)
    float cStop = entry - d * stopAtr * atr
    float hStop = isL ? math.min(entry - halfStopAtr * atr, low) : math.max(entry + halfStopAtr * atr, high)
    if halfCap
        hStop := isL ? math.max(hStop, cStop) : math.min(hStop, cStop)
    float sStop = isL ? swingLow - stopBufAtr * atr : swingHigh + stopBufAtr * atr
    float stop  = stopMode == STOP_ATR ? (half ? hStop : cStop) : isL ? math.max(sStop, cStop) : math.min(sStop, cStop)
    float risk  = d * (entry - stop)
    if risk >= syminfo.mintick and math.round_to_mintick(stop) != math.round_to_mintick(entry)
        // no-chase filter: z = distance of the signal close from the VWAP in VWAP standard deviations
        float z      = noChase and vwSd > 0 and vwBars >= 3 and sessN >= 1 ? d * (close - vwap) / vwSd : na
        bool  zNo    = noChase and not na(z) and z > noChaseSig
        // choppy session (VWAP crosses) / over-extended 1h EMA20; a skipped setup runs on invisibly like the no-chase one
        bool  chopNo = chopOn and vxRate > chopMax
        bool  emaNo  = emaOn and not na(emaSpd) and emaSpd > emaMax
        bool  ghost  = zNo or chopNo or emaNo
        if math.abs(st) == 1 and not trGhost
            evCancel := "REPLACED"
        // levels stay unrounded: the reference fills and stops on the exact values
        st      := isL ? 1 : -1
        armBar  := bar_index
        lim     := entry
        stp     := stop
        stp0    := stop
        trRisk  := risk
        trHalf  := half
        trGhost := ghost
        trSkip  := zNo ? "TOO FAR FROM VWAP" : chopNo ? "CHOPPY VWAP SESSION" : emaNo ? "1H EMA20 TOO FAST" : ""
        t1      := lim + d * tp1R * risk
        t2      := lim + d * tp2R * risk
        t3      := lim + d * tp3R * risk
        t4      := lim + d * tp4R * risk
        tF      := exitK == 1 ? t1 : exitK == 2 ? t2 : exitK == 3 ? t3 : t4
        if not ghost
            evArm  := true
            evArmL := isL
// <<< ENGINE END
