//@version=6
// =======================================================================================================
//  FRACTAL BREAKOUT GRID v3.1: 12 long breakout strategies on swing highs (fractals), with money management (strategy)
// =======================================================================================================
//  Each strategy finds swing highs on its own timeframe: a candle whose high is strictly above the Left
//  candles before it and the Right candles after it. When the last Right candle closes, the swing high
//  sets a buy stop at high x (1 + offset / 100), replacing that strategy's older buy stop. A trade opens
//  when the price rises through it; one trade at a time per strategy (a breakout while a trade is open,
//  or one the filters do not allow, is skipped); either way that buy stop is then used up.
//  Exits from the fill price: SL % and TP %. BE trigger: once the price is up this %, the SL moves to
//  the entry. Salvage trigger: once the price is down this %, the TP moves to the entry (out at
//  break-even if the price comes back). Only the first of the two happens in a trade; 0 = off.
//  Strategies 1-9: your table. 10 (5 min swing highs), 11 (15 min) and 12 (1 min): picked on gold 2021-2026.
//  v2: entry / exit options and days / hours (all off = as tested), and money: every strategy is its own
//  account (start balance, base risk) sized by one of the 28 loss-recovery rules of the Auction Retest
//  strategy, with the profit ladder, a hard cap and the leverage limit; the next trade's lots are shown.
//  v3.1: Position Size: Fixed $ risk, Fixed lots or % of equity at risk (the base the loss recovery rule sizes
//  from); a red note on the chart when a ticked strategy cannot run on the chart's timeframe.
//  The trades are worked out on the chart's candles, along open -> the nearer of high / low -> the other
//  -> close (gaps fill at the open, also on the candle where a new buy stop starts). Use a chart timeframe
//  at or below each strategy's timeframe (1 min runs all 12, 5 min all but 12). Swing highs come from
//  completed candles only, so they never repaint; on the live candle the trade marks can still change
//  until it closes (alerts are sent at the close).
//  STRATEGY BUILD: the same rules and money as the indicator, executed with TradingView orders. TradingView
//  fills an order from the candle after it is placed, so every order is placed at a candle's close for the
//  next one: the waiting buy stop (also a new one that this close confirms), the model's SL / TP, and the
//  time or weekend close due at the next open. The SL / TP sent with the buy stop are distances from the
//  fill, so a gap open above the buy stop moves them with the fill. Trade list signals: entries S1 ... S12;
//  exits TP, SL, BE (the SL was moved to the entry by the BE trigger), SALV (the TP was moved to the entry
//  by the salvage trigger), TRAIL, TIME, WKND. BE and SALV exits are at the entry price, so only the
//  commission is lost (TradingView counts them as losing trades). Where the model acts inside a candle in a
//  way TradingView cannot (an SL / TP moved by a BE, salvage or trailing trigger and hit in the same
//  candle, a dip under and back over the buy stop inside one candle, a candle-close entry), a market order
//  at the next open follows the model: "BE next open", "SALV next open", ... (exits) or "S11 next open" (an
//  entry). The smaller the chart timeframe, the fewer of these (strategy 11 on gold 2021-2026: 8.1% of its
//  trades on a 15 min chart, 5.1% on 5 min, 1.6% on 1 min). Properties: commission 0.115 per contract per
//  order = the 0.23 round-trip costs (change it with the Costs input), initial capital 120,000 (12 accounts
//  of 10,000). Run it on a 1 minute chart: all 12 strategies run there and the trades follow the rules most
//  closely (5 min: all but strategy 12).
//  Alerts: create one alert with "Any alert() function call" (sent at the candle close).
// =======================================================================================================

// ---------------------------------------------------------------- swing highs (completed candles only)
// on fractal candle k: the swing high of candle p = k - 1 - right, confirmed when candle k - 1 (= p + right) closed:
// high[p] above the highest of the left candles before it and of the right candles after it; also candle k's bar_index (expiry),
// and for the strategy's orders the swing high confirmed by candle k itself and candle k's close time
f_piv(int lft, int rgt) =>
    float lh  = ta.highest(high, lft)[rgt + 2]
    float rh  = ta.highest(high, rgt)[1]
    float p   = high[rgt + 1]
    bool  ok  = not na(lh) and not na(rh) and not na(p) and p > lh and p > rh
    // the swing high that THIS fractal candle confirms when it closes (read only at its last chart candle, when it is complete)
    float lh0 = ta.highest(high, lft)[rgt + 1]
    float rh0 = ta.highest(high, rgt)
    float p0  = high[rgt]
    bool  ok0 = not na(lh0) and not na(rh0) and not na(p0) and p0 > lh0 and p0 > rh0
    [ok ? p : na, ok ? time[rgt + 1] : na, bar_index, ok0 ? p0 : na, ok0 ? time[rgt] : na, time_close]

// trend filter: the last closed candle of the trend timeframe and its EMA (seeded with the first close)
f_ema(float src, int len) =>
    float a = 2.0 / (len + 1.0)
    var float e = na
    e := na(e) ? src : a * src + (1.0 - a) * e
    e

f_trend(int len) =>
    float e = f_ema(close, len)
    [close[1], e[1], close, e, time_close]

    float  carried = 0.0
    int    parWins = 0
    int    rdUnits = 1
    int    oscU   = 1
    float  oscP   = 0.0
    float  crL    = 0.0
    int    crN    = 0
    bool   crW    = false
    int    crP    = 1
    int    crF    = 0
    bool   crWait = false
    int    crWW   = 0
    bool   crNeedL = false
    bool   crRecT = false
    float  risk   = na
    float  units  = na
    float  bR     = na
    line   lnLvl  = na
    line   lnE    = na
    line   lnSL   = na
    line   lnTP   = na
    int    slX    = na
    int    tpX    = na
    string msg    = ""

f_px(float x) =>
    str.tostring(x, format.mintick)

f_num(float x, string f) =>
    str.tostring(x, f)

f_sgn(float x, string f) =>
    (x >= 0 ? "+" : "") + str.tostring(x, f)

f_usd(float x) =>
    (x < 0 ? "-$" : "$") + str.tostring(math.abs(x), "#,##0")

f_lots(float u) =>
    math.max(lotStep, math.round(u / lotSz / lotStep) * lotStep)

// ---------------------------------------------------------------- money (gold/recovery.py simulate, one strategy)
// the base trade's $ risk at a stop distance sd (price): Fixed $ risk, the Fixed lots' loss at that stop (costs included), or % of the account
f_base(Strat s, float sd) =>
    szMode == SZ_LOT ? baseLot * lotSz * (sd + costIn) : szMode == SZ_PCT ? math.max(0.0, s.eq) * eqPct / 100.0 : baseR

// the profit ladder now: on, the start guard (closed profit 0 or less), its base, the cushion above the lock
f_lad(Strat s, float B) =>
    bool  on = not ladOff and ladFam and s.peakNet >= ladT
    float C  = on ? math.max(0.0, s.net - ladLam * s.peakNet) : 0.0
    [on, on and s.net <= 0, on ? B + math.min((ladBmx - 1.0) * B, ladPhi * C) : B, C]

// a Crooks C - J trade at base B (rec = a recovery trade)
f_crk(Strat s, float B, bool rec, float rmin, float pf, float xr) =>
    float x = rec ? math.max(B, s.crL / s.crP * pf / xr) : B
    if rec
        if recMode == L_CRF or recMode == L_CRI
            x := math.min(x, crkFK * B)
        else if recMode == L_CRH
            x := math.min(x, math.max(B, s.eq * crkHPct / 100.0))
        else if recMode == L_CRJ
            x := math.max(B, x * math.max(0.0, 1.0 - (s.peakNet - s.net) / (crkJX * B)))
    else if s.crL > 0 and (recMode == L_CRE or recMode == L_CRI)
        x := math.min(B, rmin)
    x

// the rule's risk for the next trade (before the cap and the leverage limit), whether it is a recovery trade, and Rule C split's base
f_raw(Strat s, float sd) =>
    float bR   = f_base(s, sd)
    float xr   = s.tp / s.sl
    float rmin = eUnits * (sd + costIn)
    float add  = recMode == L_ALL ? s.recPool : recMode == L_SPLIT ? s.recPool / recN : recMode == L_PART ? s.recPool * recPct / 100.0 : recMode == L_EXACT ? s.recPool / xr : recMode == L_STEP ? recStep * s.recLosses : 0.0
    float fa = 1.0
    float fb = 1.0
    if s.recLosses >= 2
        for _k = 2 to math.min(s.recLosses, 200)
            float t2 = fa + fb
            fa := fb
            fb := t2
    float stepBase = 0.0
    if s.net >= 1000
        int kk = int(math.floor((s.net - 1000.0) / 500.0))
        stepBase := (1000.0 + 500.0 * kk) / (8.0 + kk)
    else if s.net >= 500
        stepBase := 500.0 / 5.0
    else if s.net >= 200
        stepBase := 200.0 / 3.0
    float baseC = recMode == L_CS ? math.max(bR, capOn ? math.min(stepBase, hardCap) : stepBase) : bR
    float labU  = s.lab.size() > 1 ? s.lab.first() + s.lab.last() : s.lab.first()
    int   oscNd = int(math.max(1, math.ceil((bR - s.oscP) / (bR * xr) - 1e-9)))
    bool  rec   = ((recMode == L_CRA or recMode == L_CRB) and s.crN >= 2 and s.crN % 2 == 0) or (crkFam and s.crW and s.crL > 0) or (recMode == L_CRD and s.crN >= 2 and (s.crN - 2) % 3 == 0)
    if rec and prOn and crkPr == CPR_WAIT and (s.crWait or s.crNeedL)
        rec := false
    float pf = prOn and crkPr == CPR_PCT ? crkPrPct / 100.0 : 1.0
    float r = switch recMode
        L_LAB   => bR * labU
        L_GMART => bR * math.pow(recMult, math.min(s.recLosses, 200)) + bR * s.recLosses
        L_PAR   => bR * math.pow(2.0, s.parWins)
        L_RDALE => bR * s.rdUnits
        L_DELAY => s.recLosses >= delayN ? bR + s.recPool / recN : bR
        L_EQPCT => s.eq * eqPct / 100.0
        L_OSC   => bR * math.min(s.oscU, oscNd)
        L_MART  => bR * math.pow(recMult, math.min(s.recLosses, 200))
        L_FIB   => bR * (s.recLosses == 0 ? 1.0 : fb)
        L_DALE  => bR * s.dUnits
        L_C     => math.max(baseC, s.carried / cSplit)
        L_CS    => math.max(baseC, s.carried / cSplit)
        L_CRA   => rec ? math.max(bR, s.crL / xr) : bR
        L_CRB   => rec ? math.max(bR, s.crL / cSplit / xr) : bR
        L_CRC   => rec ? math.max(bR, s.crL / s.crP * pf / xr) : bR
        L_CRD   => rec ? math.max(bR, s.crL / xr) : bR
        => bR + add
    if crkFam
        r := f_crk(s, bR, rec, rmin, pf, xr)
        [lOn, lGuard, lB, lC] = f_lad(s, bR)
        if lOn
            r := math.min(f_crk(s, lB, rec, rmin, pf, xr), r + ladGam * lC)
            if lGuard
                r := math.min(bR, rmin)
    [r, rec, baseC]

// the next trade's $ risk at fill F, shown before it happens (no state change): na = the cap would stop the strategy
f_next(Strat s, float F) =>
    float sd = F * s.sl / 100.0
    [r0, rec0, baseC] = f_raw(s, sd)
    float r = r0
    if capOn and r > hardCap
        r := capAction == CAP_STOP ? na : capAction == CAP_RESET ? math.min(baseC, hardCap) : hardCap
    if not na(r) and s.eq > 0
        r := math.min(r, s.eq * levIn * (sd + costIn) / F)
    r

// sizing at an entry (the cap's reset / stop change the state): false = the cap stops the strategy, no trade
method mmOpen(Strat s, float F) =>
    float sd = F * s.sl / 100.0
    s.bR := f_base(s, sd)
    [r0, rec0, baseC] = f_raw(s, sd)
    float r    = r0
    bool  take = true
    s.crRecT := rec0
    if capOn and r > hardCap
        if capAction == CAP_STOP
            s.stopped := true
            s.stopWhy := "stopped by the hard cap"
            take := false
        else if capAction == CAP_RESET
            s.recPool   := 0.0
            s.recLosses := 0
            s.dUnits    := 1
            s.carried   := 0.0
            s.lab.clear()
            s.lab.push(1.0)
            s.lab.push(2.0)
            s.lab.push(3.0)
            s.parWins   := 0
            s.rdUnits   := 1
            s.oscU      := 1
            s.oscP      := 0.0
            s.crL       := 0.0
            s.crN       := 0
            s.crRecT    := false
            s.crW       := false
            s.crP       := crkCn
            s.crF       := 0
            s.crWait    := false
            s.crWW      := 0
            s.crNeedL   := false
            r := math.min(baseC, hardCap)
        else
            r := hardCap
    if take and s.eq > 0
        r := math.min(r, s.eq * levIn * (sd + costIn) / F)
    s.risk  := take ? r : na
    s.units := take ? r / (sd + costIn) : na
    take

// booking a closed trade's $ result (the rule's state after the trade)
method mmBook(Strat s, float pnl) =>
    s.eq      += pnl
    s.net     += pnl
    s.pkEq    := math.max(s.pkEq, s.eq)
    s.ddU     := math.max(s.ddU, s.pkEq - s.eq)
    s.ddP     := math.max(s.ddP, (s.pkEq - s.eq) / s.pkEq)
    s.peakNet := math.max(s.peakNet, s.net)
    if s.eq <= 0
        s.stopped := true
        s.stopWhy := "account blown"
    bool win = pnl > 0
    s.carried := math.max(0.0, s.carried - pnl)
    if win
        s.recPool   := math.max(0.0, s.recPool - pnl)
        s.recLosses := s.recPool > 0 ? s.recLosses : 0
        s.dUnits    := math.max(1, s.dUnits - 1)
        s.rdUnits   += 1
        s.parWins   := s.parWins + 1 >= parN ? 0 : s.parWins + 1
        if s.lab.size() > 0
            s.lab.shift()
        if s.lab.size() > 0
            s.lab.pop()
        if s.lab.size() == 0
            s.lab.push(1.0)
            s.lab.push(2.0)
            s.lab.push(3.0)
    else
        s.recPool   += -pnl
        s.recLosses += 1
        s.dUnits    += 1
        s.rdUnits   := math.max(1, s.rdUnits - 1)
        s.parWins   := 0
        s.lab.push(s.lab.size() > 1 ? s.lab.first() + s.lab.last() : s.lab.first())
    s.oscP += pnl
    if s.oscP >= s.bR
        s.oscU := 1
        s.oscP := 0.0
    else if win
        s.oscU += 1
    bool lastPart = s.crP == 1
    s.crL := math.max(0.0, s.crL - pnl)
    if recMode == L_CRG
        [gOn, gGuard, gB, gC] = f_lad(s, s.bR)
        s.crL := math.min(s.crL, crkGX * gB)
    if s.crL < 0.01 or ((recMode == L_CRA or recMode == L_CRD) and win and s.crRecT)
        s.crL     := 0.0
        s.crN     := 0
        s.crP     := crkCn
        s.crF     := 0
        s.crWait  := false
        s.crWW    := 0
        s.crNeedL := false
    else
        s.crN += 1
        s.crP := not win ? crkCn : s.crRecT ? math.max(1, s.crP - 1) : s.crP
        if prOn
            if s.crWait
                s.crWW := win ? s.crWW + 1 : 0
                if s.crWW >= crkPrM
                    s.crWait  := false
                    s.crWW    := 0
                    s.crNeedL := true
            else if s.crNeedL and not win
                s.crNeedL := false
            if s.crRecT and win
                s.crF := 0
                if crkPr == CPR_PCT and lastPart
                    s.crL := 0.0
                    s.crN := 0
                    s.crP := crkCn
            else if s.crRecT
                s.crF += 1
                if s.crF >= crkPrN
                    s.crF := 0
                    if crkPr == CPR_FORGET or crkPr == CPR_PCT
                        s.crL := 0.0
                        s.crN := 0
                        s.crP := crkCn
                    else if crkPr == CPR_WAIT
                        s.crWait  := true
                        s.crWW    := 0
                        s.crNeedL := false
    s.crW := win
    true

// ---------------------------------------------------------------- trades
// a trade opens at px (false: the hard cap's Stop trading refused it)
method openAt(Strat s, float px, bool gap) =>
    bool counted = time >= statsFrom
    bool take    = true
    if mmOn and counted
        take := s.mmOpen(px)
    if take
        s.pos   := true
        s.F     := px
        s.SLp   := px * (1.0 - s.sl / 100.0)
        s.TPp   := px * (1.0 + s.tp / 100.0)
        s.BEl   := px * (1.0 + s.be / 100.0)
        s.SVl   := px * (1.0 - s.sv / 100.0)
        s.beOn  := false
        s.svOn  := false
        s.trOn  := false
        s.hi    := px
        s.eTime := time
        s.cnt   := mmOn and counted
        s.dCnt  += 1
        if showTr
            s.lnE  := line.new(bar_index, px, bar_index, px, color = s.col, width = 2)
            s.lnTP := line.new(bar_index, s.TPp, bar_index, s.TPp, color = C_TP)
            s.lnSL := line.new(bar_index, s.SLp, bar_index, s.SLp, color = C_SL)
            s.slX  := bar_index
            s.tpX  := bar_index
        string mTxt = s.cnt ? " | " + f_num(f_lots(s.units), "0.00##") + " lot, risk " + f_usd(s.risk) : ""
        if showLbl
            label.new(bar_index, px, str.tostring(s.id) + (gap ? " BUY (gap)" : " BUY") + (s.cnt ? " " + f_num(f_lots(s.units), "0.00##") : ""), color = s.col, textcolor = color.white, style = label.style_label_up, size = size.tiny)
        if alEnt
            s.msg += "FBG S" + str.tostring(s.id) + " BUY " + f_px(px) + (gap ? " (gap open)" : "") + " | SL " + f_px(s.SLp) + " | TP " + f_px(s.TPp) + mTxt + "\n"
    take

method closeAt(Strat s, float px, int rsn) =>
    float pnl = px - s.F - costIn
    float pct = pnl / s.F * 100.0
    float rr  = pnl / (s.F * s.sl / 100.0)
    float usd = s.cnt ? s.units * pnl : na
    if s.eTime >= statsFrom
        s.n      += 1
        s.sumPct += pct
        s.sumR   += rr
        if pnl > 0
            s.nW += 1
            s.gw += rr
        else
            s.gl -= rr
        if rsn == 2
            s.st  += 1
            s.mst := math.max(s.mst, s.st)
        else if rsn == 1
            s.st := 0
        s.cum += rr
        s.pk  := math.max(s.pk, s.cum)
        s.dd  := math.max(s.dd, s.pk - s.cum)
        if rsn == 1
            s.nTP += 1
        else if rsn == 2
            s.nSL += 1
        else if rsn == 3
            s.nBE += 1
        else if rsn == 4
            s.nSV += 1
        else
            s.nOth += 1
    if s.cnt
        s.mmBook(usd)
    string rs = rsn == 1 ? "TP" : rsn == 2 ? "SL" : rsn == 3 ? "BE" : rsn == 4 ? "SALV" : rsn == 5 ? "TRAIL" : rsn == 6 ? "TIME" : "WKND"
    s.xWhy := rs
    s.xBar := bar_index
    if not na(s.lnE)
        s.lnE.set_x2(bar_index)
    if not na(s.lnSL)
        s.lnSL.set_x2(bar_index)
    if not na(s.lnTP)
        s.lnTP.set_x2(bar_index)
    s.lnE  := na
    s.lnSL := na
    s.lnTP := na
    if showLbl
        label.new(bar_index, px, str.tostring(s.id) + " " + rs + " " + f_sgn(pct, "0.00") + "%" + (s.cnt ? " " + (usd >= 0 ? "+" : "") + f_usd(usd) : ""), color = pnl > 0 ? C_WIN : C_LOSS, textcolor = color.white, style = label.style_label_down, size = size.tiny)
    if alExt
        s.msg += "FBG S" + str.tostring(s.id) + " EXIT " + rs + " " + f_px(px) + " (" + f_sgn(pct, "0.00") + "% after costs, " + f_sgn(rr, "0.00") + "R" + (s.cnt ? ", " + (usd >= 0 ? "+" : "") + f_usd(usd) : "") + ")\n"
    s.pos := false
    s.cnt := false
    true

// the SL / TP line continues at a new level (one new line per candle at most; the candle where each line starts is kept)
method moveSL(Strat s, float lvl) =>
    if not na(s.lnSL)
        if na(s.slX) or bar_index > s.slX
            s.lnSL.set_x2(bar_index)
            s.lnSL := line.new(bar_index, lvl, bar_index, lvl, color = C_MOVE, style = line.style_dotted, width = 2)
            s.slX  := bar_index
        else
            s.lnSL.set_y1(lvl)
            s.lnSL.set_y2(lvl)
    true

method moveTP(Strat s, float lvl) =>
    if not na(s.lnTP)
        if na(s.tpX) or bar_index > s.tpX
            s.lnTP.set_x2(bar_index)
            s.lnTP := line.new(bar_index, lvl, bar_index, lvl, color = C_MOVE, style = line.style_dotted, width = 2)
            s.tpX  := bar_index
        else
            s.lnTP.set_y1(lvl)
            s.lnTP.set_y2(lvl)
    true

// BE trigger reached: the SL moves to the entry (never down)
method doBE(Strat s) =>
    s.beOn := true
    if s.F > s.SLp
        s.SLp  := s.F
        s.moveSL(s.F)
    true

// salvage trigger reached: the TP moves to the entry
method doSV(Strat s) =>
    s.svOn := true
    s.TPp  := s.F
    s.moveTP(s.F)
    true

// trailing stop: once the highest price is the start % above the fill, the SL follows it at the distance %
method trail(Strat s) =>
    if trD > 0 and s.hi >= s.F * (1.0 + trA / 100.0)
        float tl = s.hi * (1.0 - trD / 100.0)
        if tl > s.SLp
            s.SLp  := tl
            s.trOn := true
            s.moveSL(tl)
    true

// the waiting buy stop is used up (filled, skipped or expired)
method useLvl(Strat s) =>
    s.lvlOn := false
    if not na(s.lnLvl)
        s.lnLvl.set_x2(bar_index)
    s.lnLvl := na
    true

// may a breakout open a trade now
method canEnter(Strat s, bool okb) =>
    okb and not s.pos and not s.stopped and (maxDay <= 0 or s.dCnt < maxDay)

// the price moves up from x to y: TP, BE trigger and the buy stop, lowest first (equal: TP, BE, buy stop)
method up(Strat s, float x, float y, bool okb) =>
    for _i = 0 to 7
        float best = na
        int   ev   = 0
        if s.pos
            if x < s.TPp and s.TPp <= y
                best := s.TPp
                ev   := 1
            if s.be > 0 and not s.beOn and not s.svOn and x < s.BEl and s.BEl <= y and (na(best) or s.BEl < best)
                best := s.BEl
                ev   := 2
        if entMode == E_STOP and s.lvlOn and x < s.E and s.E <= y and (na(best) or s.E < best)
            best := s.E
            ev   := 3
        if ev == 0
            break
        if ev == 1
            s.closeAt(s.TPp, s.svOn ? 4 : 1)
        else if ev == 2
            s.doBE()
        else
            if s.canEnter(okb)
                s.openAt(s.E, false)
            s.useLvl()
    if s.pos and y > s.hi
        s.hi := y
    true

// the price moves down from x to y: trailing update, then SL and salvage trigger, highest first (equal: SL)
method dn(Strat s, float x, float y) =>
    if s.pos
        s.trail()
    for _i = 0 to 7
        float best = na
        int   ev   = 0
        if s.pos
            if y <= s.SLp and s.SLp < x
                best := s.SLp
                ev   := 1
            if s.sv > 0 and not s.svOn and not s.beOn and y <= s.SVl and s.SVl < x and (na(best) or s.SVl > best)
                best := s.SVl
                ev   := 2
        if ev == 0
            break
        if ev == 1
            s.closeAt(s.SLp, s.trOn ? 5 : s.beOn ? 3 : 2)
        else
            s.doSV()
    true

method seg(Strat s, float x, float y, bool okb) =>
    if y > x
        s.up(x, y, okb)
    else if y < x
        s.dn(x, y)
    true

// one chart candle: a new swing high, expiry, the open (trail, exits, BE / salvage, then a gap fill), the path, a close entry
method onBar(Strat s, float ph, int pt, int hb, float pc, bool okb, bool wk, int dno) =>
    s.msg := ""
    if na(s.dPrev) or dno != s.dPrev
        s.dCnt  := 0
        s.dPrev := dno
    bool lvlNew = false
    if not na(ph) and not na(pt) and (na(s.lvlT) or pt != s.lvlT)
        if not na(s.lnLvl)
            s.lnLvl.set_x2(bar_index)
        s.lvlT  := pt
        s.lvlOn := true
        s.lvlH  := ph
        s.E     := ph * (1.0 + s.off / 100.0)
        s.kAct  := hb
        s.armed := false
        lvlNew  := true
        s.lnLvl := showLvl ? line.new(bar_index, s.E, bar_index, s.E, color = s.col, style = line.style_dashed) : na
        if alLvl
            float nr = na
            if mmOn and not s.stopped and time >= statsFrom
                nr := f_next(s, s.E)
            s.msg += "FBG S" + str.tostring(s.id) + " BUY STOP " + f_px(s.E) + " (swing high " + f_px(ph) + ", " + s.tf + " " + str.tostring(s.r) + "/" + str.tostring(s.l) + ") | SL " + f_num(s.sl, "0.0##") + "% | TP " + f_num(s.tp, "0.0##") + "%" + (na(nr) ? "" : " | " + f_num(f_lots(nr / (s.E * s.sl / 100.0 + costIn)), "0.00##") + " lot, risk " + f_usd(nr)) + "\n"
    if s.lvlOn and expN > 0 and not na(s.kAct) and hb - s.kAct >= expN
        s.useLvl()
    if s.pos
        s.trail()
        int xr = 0
        if open >= s.TPp
            xr := s.svOn ? 4 : 1
        else if open <= s.SLp
            xr := s.trOn ? 5 : s.beOn ? 3 : 2
        else if tExitH > 0 and time >= s.eTime + math.round(tExitH * 3600000)
            xr := 6
        else if wk
            xr := 7
        if xr > 0
            s.closeAt(open, xr)
        else
            if s.be > 0 and not s.beOn and not s.svOn and open >= s.BEl
                s.doBE()
            else if s.sv > 0 and not s.svOn and not s.beOn and open <= s.SVl
                s.doSV()
            if open > s.hi
                s.hi := open
    if entMode == E_STOP and s.lvlOn and not na(pc) and pc < s.E and open >= s.E
        if s.canEnter(okb)
            s.openAt(open, true)
        s.useLvl()
    bool  hf = (high - open) < (open - low)
    float a  = hf ? high : low
    float b  = hf ? low : high
    s.seg(open, a, okb)
    s.seg(a, b, okb)
    s.seg(b, close, okb)
    if entMode == E_CLOSE and s.lvlOn
        if s.armed and close >= s.E
            if s.canEnter(okb)
                s.openAt(close, false)
            s.useLvl()
        else if close < s.E
            s.armed := true
    if s.lvlOn and not na(s.lnLvl)
        s.lnLvl.set_x2(bar_index)
    if s.pos
        if not na(s.lnE)
